    function crearDesarrollador(nombre, lenguaje) {
      return {
        nombre: nombre,
        lenguaje: lenguaje,
        saludar: function() {
          console.log("Hola, soy " + this.nombre + " y programo en " + this.lenguaje);
        }
      };
    }

    let desarrolladorSamuel = crearDesarrollador("Samuel", "JavaScript");
    desarrolladorSamuel.saludar();

    // Declaramos la variable y la utilizamos
    let numeroProyectos = 0;
    console.log("Proyectos al inicio:", numeroProyectos);

    // Actualizamos el número de proyectos
    numeroProyectos = numeroProyectos + 3; 
    console.log("Ahora hay", numeroProyectos, "proyectos en desarrollo.");

    // Definimos la función antes de usarla
    function iniciarFabricaDesarrollo() {
      console.log("Fábrica de desarrollo iniciada...");
    }
    iniciarFabricaDesarrollo();

    console.log("Este es un proyecto sencillo con Samuel para una fábrica de desarrollo (versión con uso de la variable).");
  </script>
